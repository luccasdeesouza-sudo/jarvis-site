const $ = id => document.getElementById(id);
function el(tag, attrs){ const e = document.createElement(tag); if(attrs) Object.assign(e, attrs); return e; }
function escapeHtml(s){ return String(s).replace(/[&<>"'`]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;','`':'&#96;'}[c])); }
async function sha256(msg){ const enc = new TextEncoder(); const data = enc.encode(msg); const hash = await crypto.subtle.digest('SHA-256', data); return Array.from(new Uint8Array(hash)).map(b=>b.toString(16).padStart(2,'0')).join(''); }

/* ---------- ENSURE ADMIN ---------- */
async function ensureAdmin(){
  try{
    const snap = await db.ref('users/' + ADMIN_USER).once('value');
    if(!snap.exists()){
      await db.ref('users/' + ADMIN_USER).set({
        passwordHash: ADMIN_HASH,
        email: ADMIN_EMAIL,
        approved: true,
        isAdmin: true,
        createdAt: Date.now()
      });
      console.log('Admin lux criado no DB.');
    } else {
      console.log('Admin já existe.');
    }
  }catch(e){
    console.warn('Erro ensureAdmin (verifique firebaseConfig):', e);
  }
}

    if(snap.exists()){ alert('Usuário já existe'); return; }
    if(p.length < 6){ alert('Senha mínima 6 caracteres'); return; }
    const h = await sha256(p);
    await db.ref('users/' + u).set({ passwordHash: h, email: email || '', approved: false, isAdmin: false, createdAt: Date.now() });
    alert('Solicitação enviada: aguarde aprovação do Lux.');

    // try username
    let userKey = null, userVal = null;
    const direct = await db.ref('users/' + id).once('value');
    if(direct.exists()){ userKey = id; userVal = direct.val(); }

/* ---------- USERS LIST / PENDING LISTeners ---------- */
function listenUsers(){
  db.ref('users').on('value', snap => {
    const data = snap.val() || {};
    renderApproved(data);
    renderPending(data);
  });
}

function renderApproved(data){
  const out = $('approvedList'); out.innerHTML = '';
  Object.keys(data).filter(k => data[k].approved).sort().forEach(u=>{
    const d = data[u];
    const div = el('div'); div.style.padding = '6px 4px';
    div.innerHTML = `<strong>${escapeHtml(u)}</strong><div style="color:#9f9; font-size:12px">${escapeHtml(d.email||'')}</div>`;
    out.appendChild(div);
  });
}

function renderPending(data){
  const out = $('pendingList'); out.innerHTML = '';
  Object.keys(data).filter(k => !data[k].approved).sort().forEach(u=>{
    const d = data[u];
    const item = el('div', {}); item.className = 'pendingItem';
    const left = el('div'); left.innerHTML = `<strong>${escapeHtml(u)}</strong><div style="font-size:12px;color:#9f9">${new Date(d.createdAt||Date.now()).toLocaleString()}</div>`;
    const right = el('div'); 
    const yes = el('button'); yes.textContent='Sim'; yes.style.marginRight='6px';
    const no = el('button'); no.textContent='Não'; no.className='ghost';
    yes.addEventListener('click', async ()=>{ await db.ref('users/' + u + '/approved').set(true); alert(u + ' aprovado.'); });
    no.addEventListener('click', async ()=>{ await db.ref('users/' + u).remove(); alert(u + ' rejeitado.'); });
    right.appendChild(yes); right.appendChild(no);
    item.appendChild(left); item.appendChild(right);
    out.appendChild(item);
  });
}

    $('adminPanel').classList.remove('hidden');

  } else {

/* ---------- CHAT functions (realtime) ---------- */
async function openChatWindow(){
  // show chat window
  $('chatWindow').style.display = 'flex';
  // attach messages listener to /messages/general
  if(messagesRef) messagesRef.off();
  messagesRef = db.ref('messages/general');
  messagesRef.limitToLast(200).once('value', snap=>{
    const data = snap.val() || {};
    $('chatBody').innerHTML = '';
    Object.keys(data).forEach(k => appendMsg(data[k]));
    $('chatBody').scrollTop = $('chatBody').scrollHeight;
  });
  messagesRef.limitToLast(1).on('child_added', snap => {
    appendMsg(snap.val());
  });
}

function closeChatWindow(){
  $('chatWindow').style.display = 'none';
  if(messagesRef) messagesRef.off();
}

function appendMsg(m){
  const body = $('chatBody');
  const d = el('div');
  d.className = 'msg' + (currentUser && m.from === currentUser.username ? ' me' : '');
  d.innerHTML = `<div style="font-size:12px;color:#040">${escapeHtml(m.from)} • ${new Date(m.at).toLocaleTimeString()}</div><div>${escapeHtml(m.text)}</div>`;
  body.appendChild(d);
  body.scrollTop = body.scrollHeight;
}

async function sendMessage(){
  if(!currentUser){ alert('Faça login antes de enviar.'); return; }
  const text = $('chatText').value.trim();
  if(!text) return;
  const msg = { from: currentUser.username, text, at: Date.now() };
  await db.ref('messages/general').push(msg);
  $('chatText').value = '';
}
